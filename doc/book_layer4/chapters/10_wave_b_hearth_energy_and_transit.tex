\subsection{Integer WASM CVRP, Battery-Swap Leases, Cold-Chain, and Multi-Hop HTLCs}

Route consolidation inside \texttt{col-execd} executes as a deterministic, float-free
WebAssembly module solving the \textbf{Integer Capacitated Vehicle Routing Problem (\texttt{CVRP})}
via Clarke-Wright parallel savings and 2-opt edge exchange in exact millimeters (\texttt{mm}),
milligrams (\texttt{cargo\_mass\_mg}), milliliters (\texttt{volume\_ml}), and microjoules
(\texttt{traction\_uj}). Candidate routes must satisfy four invariants:
\begin{enumerate}
    \item \textbf{Thermodynamic Efficiency \& Detour Bounds:} Delivered effective exergy
    ratio $\eta_{\mathrm{route}}(\mathcal{P}) = \lfloor (\mathcal{E}_{\mathrm{eff}} \cdot 10{,}000) / \mathcal{E}_0 \rfloor \ge 6000\,\mathrm{bp}$
    ($60\%$), passenger carpool detour $\le 3000\,\mathrm{bp}$ ($30\%$), and zero intersection
    with BPVP \texttt{No-Transit} vetoed glades or salmon corridors.
    \item \textbf{Depot Battery-Swap Reservation Leases:} Before an e-trike departs on a
    multi-hop watershed leg ($A \to B \to C$), \texttt{wf.beta.cvrp\_milk\_run} acquires an
    HLC-bounded lexicographic reservation (\texttt{lease:depot:B:batt\_pack}) on a docked
    $48\,\mathrm{V}$ $\mathrm{LiFePO_4}$ pack verified at $\mathrm{SoC} \ge 9000\,\mathrm{bp}$
    at intermediate depot $B$, guaranteeing zero mid-route stranding.
    \item \textbf{In-Transit Cold-Chain \& Hot-Tote Thermal Envelopes:} Smart-containers
    carrying Scenario Gamma dairy/meals or Scenario Chi/Delta pasteurized donor milk/biologics
    carry continuous Layer 2 thermocouple envelopes ($[275{,}150, 277{,}150]\,\mathrm{mK}$
    cold chain or $\ge 336{,}150\,\mathrm{mK}$ hot tiffins). Any excursion beyond safe
    thermal-time bounds fires a \textbf{Regime 2 \texttt{ColdChainBreachSaga}}, locking the
    recipient food/clinical latch upon arrival and diverting organic mass to BSFL/compost
    while alerting Scenario Chi/Gamma to re-dispatch from backup stock.
    \item \textbf{Multi-Hop BPVP HTLC Handoffs \& Conduit Plausibility:} Across intermediate
    depots $A \to B \to C$, Layer 3 locks chained Hashed Timelock Contracts
    $\mathrm{HTLC}_i = \langle H(S), \Delta t_i, Q_{\mathrm{uj}}, u_i, v_i \rangle$ with
    strictly decreasing timelocks ($\Delta t_{A\to B} > \Delta t_{B\to C} + \Delta t_{\mathrm{margin}}$).
    Destination $C$ reveals preimage $S$ only when its Layer 2 load-cell scale and BLE
    challenge sign $\sigma_{\mathrm{delivery}}$---atomically settling $B \leftrightarrow C$
    and $A \leftrightarrow B$ while preserving $B_{u\to v} + B_{v\to u} = 0$. Simultaneously,
    Artery 4 cross-checks pumped water flow against motor exergy
    ($\Delta E_{\mathrm{pump\_uj}} \ge \lfloor \rho g Q_{\mathrm{water}} h_{\mathrm{head}} / \eta_{\mathrm{motor}} \rfloor$)
    and physical RPZ/air-gap isolation ($\Delta h \ge 2 D_{\mathrm{pipe}}$), quarantining
    spoofed flow sensors automatically.
\end{enumerate}

\begin{lstlisting}[language=C++,caption={Float-free WASM CVRP leg, battery-swap lease, cold-chain check, and multi-hop HTLC gate in Scenario Beta.},label={lst:beta-cvrp-htlc}]
struct alignas(8) BetaCvrpAndHtlcLeg {
    uint64_t payload_mass_mg;           // Must be <= vehicle max_payload_mg
    uint64_t max_payload_mg;
    uint64_t traction_energy_uj;        // Mechanical work m*g*dh/eta + m*Crr*d
    uint64_t effective_delivered_uj;    // Required eta_route >= 6000 bp (60%)
    uint32_t upstream_htlc_timeout_ms;  // Delta t_(A->B)
    uint32_t downstream_htlc_timeout_ms;// Delta t_(B->C); requires strict margin
    uint32_t tote_temp_mk;              // Cold chain [275150, 277150] mK
    uint16_t swap_pack_reserved_soc_bp; // Docked 48V pack at relay depot >= 9000 bp
    uint16_t carpool_detour_bp;         // Passenger detour <= 3000 bp
    uint8_t  hits_no_transit_glade;     // 0 = clear; 1 = vetoed by L5 BPVP
    uint8_t  cold_chain_required;       // 1 = dairy / donor milk / clinical payload
};

[[nodiscard]] constexpr bool ValidateBetaMultiHopDispatch(const BetaCvrpAndHtlcLeg& leg) noexcept {
    if (leg.hits_no_transit_glade != 0u || leg.payload_mass_mg > leg.max_payload_mg) return false;
    if (leg.carpool_detour_bp > 3000u || leg.swap_pack_reserved_soc_bp < 9000u) return false;
    if (leg.upstream_htlc_timeout_ms <= leg.downstream_htlc_timeout_ms + 600000u) return false;
    const uint64_t total_uj = leg.effective_delivered_uj + leg.traction_energy_uj;
    if (total_uj == 0ULL || (leg.effective_delivered_uj * 10000ULL) / total_uj < 6000ULL) {
        return false;
    }
    if (leg.cold_chain_required != 0u) {
        if (leg.tote_temp_mk < 275150u || leg.tote_temp_mk > 277150u) return false;
    }
    return true;
}
\end{lstlisting}
