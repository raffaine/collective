In Equation~\eqref{eq:zero-canary}, the static binary auditor scans all WASM and ELF code sections to verify zero

\subsection{Dormant Equilibrium (``The \TeX{} Asymptote''), Universal Civic Baseline $V_i(t) = 1 + R_i(t)$, and Substrate Drift Wake-Up (\texttt{D2}, \texttt{D3})}

    R_i(t) = R_i(t_0)\,e^{-\lambda(t - t_0)} + \int_{t_0}^{t} \dot{r}_i(\tau)\,e^{-\lambda(t - \tau)}\,\mathrm{d}\tau \;\xrightarrow{\;\dot{r}_i \to 0\;}\; 0 \qquad (\forall\,i \in \mathcal{M}_{\mathrm{node}}).

\end{equation}
The working group enters \textbf{Dormant Equilibrium (``The \TeX{} Asymptote'')}, existing as silent public

\begin{equation}
    V_i(t) = B + R_i(t) = 1 + R_i(t), \qquad V_{i,\mathrm{bp}}(t) = 10{,}000 + R_{i,\mathrm{bp}}(t) \qquad (\forall\,i \in \{1,\dots,N_s\}).

\end{equation}
