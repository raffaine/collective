\subsection{Fixed-Point Integer BKT, Co-Claim Interlocks, and 3-of-3 Capstones}

To track procedural mastery inside \texttt{col-execd} without IEEE-754 floats, Scenario Kappa
implements exact \textbf{Fixed-Point Integer Bayesian Knowledge Tracing (\texttt{BKT})}
\parencite{corbett1995bkt} over basis points $[0, 10{,}000]\,\mathrm{bp}$.

\begin{definition}[Fixed-Point Integer BKT, Supervised Co-Claim, and 3-of-3 Capstone]
\label{def:kappa-bkt}
Let $L_t = \mathtt{p\_known\_bp} \in [0, 10{,}000]$, $T = \mathtt{p\_learn\_bp}$,
$S = \mathtt{p\_slip\_bp}$, and $G = \mathtt{p\_guess\_bp}$ be basis-point parameters
evaluated in 64-bit unsigned intermediates. Given observation $o_t \in \{0, 1\}$ on a
supervised rubric step, the posterior mastery $L'_{t,\mathrm{bp}}$ and transition update
$L_{t+1,\mathrm{bp}}$ are:
\begin{align}
    L'_{t,\mathrm{bp}}(1) &= \left\lfloor \frac{L_t (10{,}000 - S) \cdot 10{,}000}{L_t (10{,}000 - S) + (10{,}000 - L_t) G} \right\rfloor, \quad
    L'_{t,\mathrm{bp}}(0) = \left\lfloor \frac{L_t S \cdot 10{,}000}{L_t S + (10{,}000 - L_t)(10{,}000 - G)} \right\rfloor, \label{eq:kappa-bkt-post}\\
    L_{t+1,\mathrm{bp}} &= L'_{t,\mathrm{bp}}(o_t) + \left\lfloor \frac{\big(10{,}000 - L'_{t,\mathrm{bp}}(o_t)\big) \cdot T}{10{,}000} \right\rfloor. \label{eq:kappa-bkt-step}
\end{align}
High-hazard equipment interlocks and credential minting enforce three gates:
\begin{enumerate}
    \item \textbf{Supervised Co-Claim Interlock (\texttt{L0} or Rusty \texttt{L1+}):} Layer 2

    $L_{t,\mathrm{bp}} \ge 9500\,\mathrm{bp}$ ($95\%$), supervised hours met, physical
    capstone metrology error $\le \epsilon_{\max\,\mu\mathrm{m}}$, and unannounced

\end{definition}

\begin{lstlisting}[language=C++,caption={Exact 64-bit integer Bayesian Knowledge Tracing step and Supervised Co-Claim hardware interlock.},label={lst:kappa-bkt-coclaim}]
[[nodiscard]] constexpr uint16_t StepIntegerBktBasisPoints(
    uint16_t p_known_bp, uint16_t p_learn_bp, uint16_t p_slip_bp, uint16_t p_guess_bp, bool success
) noexcept {
    const uint64_t L = p_known_bp, T = p_learn_bp, S = p_slip_bp, G = p_guess_bp;
    uint64_t num = 0ULL, den = 0ULL;
    if (success) {

        den = L * (10000ULL - S) + (10000ULL - L) * G;

        num = L * S * 10000ULL;
        den = L * S + (10000ULL - L) * (10000ULL - G);
    }
    const uint64_t post_bp = (den > 0ULL) ? (num / den) : L;
    const uint64_t next_bp = post_bp + ((10000ULL - post_bp) * T) / 10000ULL;
    return static_cast<uint16_t>(next_bp > 10000ULL ? 10000ULL : next_bp);
}

}
